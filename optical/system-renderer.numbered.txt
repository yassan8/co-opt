 2813: /**

 2815:  * @param {Array} opticalSystemRows - Optical system data

 2820:     if (!opticalSystemRows || opticalSystemRows.length === 0) {
 2821:         return null;

 2824:     const DEBUG_STOP = !!(typeof globalThis !== 'undefined' && globalThis.__COOPT_DEBUG_STOP_SURFACE);

 2826:         // 蜈牙ｭｦ邉ｻ繝・・繧ｿ蜈ｨ菴薙ｒ繝・ヰ繝・げ蜃ｺ蜉・

 2840:         const objTypeRaw = surface['object type'] ?? surface.object ?? surface.objectType;
 2841:         const objTypeNorm = String(objTypeRaw ?? '').trim().toUpperCase();

 2843:             // console.log(`識 [findStopSurface] Stop髱｢逋ｺ隕・ Surface ${i}`);

 2863:                         stopZ += parseFloat(thickness) || 0;

 2868:             // stopZ縺梧焚蛟､縺ｧ縺ゅｋ縺薙→繧堤｢ｺ隱・

 2878:                 'semidia',          // 螳滄圀縺ｮ繝輔ぅ繝ｼ繝ｫ繝牙錐・・
 2879:                 'semiDiameter', 'semi-diameter', 'semi_diameter',

 2881:                 'semiDia', 'aper', 'halfDiameter', 'half-diameter',
 2882:                 'Clear_Aperture', 'clearAperture', 'clear_aperture'
 2883:             ];
 2884: 
 2885:             const radiusSources = [
 2886:                 surface,
 2887:                 surface?.parameters,
 2888:                 surface?.aperture,
 2889:                 surface?.variables,
 2890:             ];
 2891: 
 2892:             const readNumericLike = (raw) => {
 2893:                 if (raw === undefined || raw === null || raw === '') return null;
 2894:                 if (typeof raw === 'number') return Number.isFinite(raw) ? raw : null;
 2895:                 if (typeof raw === 'string') {
 2896:                     const n = parseFloat(raw);
 2897:                     return Number.isFinite(n) ? n : null;
 2898:                 }
 2899:                 if (typeof raw === 'object') {
 2900:                     const nested = raw?.value;
 2901:                     if (nested === undefined || nested === null || nested === '') return null;
 2902:                     const n = parseFloat(nested);
 2903:                     return Number.isFinite(n) ? n : null;
 2904:                 }
 2905:                 return null;

 2907:

 2909:             for (const field of radiusFields) {
 2910:                 let resolved = null;
 2911:                 for (const src of radiusSources) {
 2912:                     resolved = readNumericLike(src?.[field]);
 2913:                     if (resolved !== null) break;
 2914:                 }
 2915:                 // console.log(`  ${field}: ${resolved} (resolved)`);
 2916:                 if (resolved !== null) {

 2919:                     break;
 2920:                 }
 2921:             }
 2922: 
 2923:             // Stop block parameter name precedence: parameters.semiDiameter
 2924:             if ((!Number.isFinite(stopRadius) || stopRadius <= 0) && surface?.parameters) {

 2928:                 }
 2929:             }
 2930: 
 2931:             // Block variable fallback: variables.semiDiameter.value
 2932:             if ((!Number.isFinite(stopRadius) || stopRadius <= 0) && surface?.variables) {
 2933:                 const fallbackVarSemiDiameter = readNumericLike(surface.variables.semiDiameter);

 2935:                     stopRadius = fallbackVarSemiDiameter;

 2937:             }
 2938:

 2940:                 for (const src of radiusSources) {

 2943:                         if (!/semi\s*dia|semidia|aperture|clear[_\s-]*aperture/i.test(String(key))) continue;
 2944:                         const candidate = readNumericLike(src[key]);
 2945:                         if (candidate !== null && candidate > 0) {
 2946:                             stopRadius = candidate;
 2947:                             break;
 2948:                         }
 2949:                     }
 2950:                 }
 2951:                 if (!Number.isFinite(stopRadius) || stopRadius <= 0) {
 2952:                     stopRadius = 10;
 2953:                 }
 2954:             }
 2955:             
 2956:             // 謇句虚縺ｧ險ｭ螳壹＆繧後◆蜊雁ｾ・､縺後≠繧九°繝√ぉ繝・け

 3069:             }
 3070:         }
